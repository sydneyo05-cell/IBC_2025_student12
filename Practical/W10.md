# Load data
toads <- read.csv("Toad_Handedness.tsv", sep="\t")

nExp <- 100000 # set values for simulation 
toads$p <- NA

for(t in 1:nrow(toads)){
  right_ct <- c()

  for(i in 1:nExp){
    experiment <- sample(c("R","L"), size=toads$Trials[t], replace=T)
    right_ct[i] <- sum(experiment=="R")
  }

  toads$p[t] <- mean(abs(right_ct-toads$Trials[t]/2) >=
                     abs(toads$Right[t]-toads$Trials[t]/2))
}

# Determine handedness
toads$Hand <- "None"
toads$Hand[toads$p<0.05 & toads$Right>toads$Trials/2] <- "Right"
toads$Hand[toads$p<0.05 & toads$Right<toads$Trials/2] <- "Left"
