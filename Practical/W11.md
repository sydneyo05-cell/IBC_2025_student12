```R
sleep <- c(30.5, 30.6, 30.4, 27.2, 33.4, 34.5, #Data from the National Health Interview Survey, 2024
           27.2, 27.9, 40.2, 28.9, 29.0)

groups <- c("Total", "Men", "Women", "18-34", #Names for each group
            "35-49", "50-64", "65 and older",
            "Asian, non-Hispanic", "Black, non-Hispanic",
            "White, non-Hispanic", "Hispanic")

colors <- c(rep("steelblue4", 3),  #Colors for the different groups
            rep("olivedrab3", 4),
            rep("darkseagreen3", 4))

barplot(rev(sleep),   #Create horizontal barplot
        names=rev(groups),
        horiz=TRUE,
        las=1,
        col=rev(colors),
        border=NA,
        xlim=c(0, 50),
        xlab="Percent",
        main="Adults Getting Less Than 7 Hours of Sleep")
```
