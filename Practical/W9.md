files <- list.files("R/DataFiles/Saavedra2013",
                   full.names = TRUE)
results <- data.frame()

for (file in files) {

  mat <- as.matrix(read.table(file))

  rows <- nrow(mat)
  cols <- ncol(mat)
  p_1 <- sum(mat == 1) / length(mat)
  p_0 <- sum(mat == 0) / length(mat)

  results <- rbind(results, data.frame(
    File = basename(file),
    Nrow = rows,
    Ncol = cols,
    p_1 = p_1,
    p_0 = p_0
  ))
}
write.table(results, "MatrixStats.tsv",
            sep = "\t", row.names = FALSE)

print(results)
