## 1. Are most toads right handed?

### 1

# Load data
toads <- read.csv("Toad_Handedness.tsv", sep="\t")

nExp <- 100000
toads$p <- NA
toads$Hand <- "None"

for(t in 1:nrow(toads)){

  right_ct <- c()

  for(i in 1:nExp){
    experiment <- sample(c("R","L"),
                         size=toads$N_trials[t],
                         prob=c(0.5,0.5), replace=T)

    right_ct[i] <- sum(experiment=="R")
  }

  # Two-sided p-value
  expected <- toads$N_trials[t]/2
  observed <- toads$N_rights[t]

  toads$p[t] <- sum(abs(right_ct-expected) >=
                     abs(observed-expected))/nExp

  if(toads$p[t] < 0.05){
    if(observed > expected){
      toads$Hand[t] <- "Right"
    } else {
      toads$Hand[t] <- "Left"
    }
  }
}

print(toads)
